	name:Research Price

*rsTime
	name:Research Time
	on start:this is 60
	hidden
*rsTimer
	name:Research Timer
	on tick:
		if (this = 0)
			yield 1 rsPoint
			this is -1
		else if (this > 0) 
			lose 1 this
		end
	end

	hidden

*frenzySecond|frenzySeconds 
	name:Frenzy!
	text:Frenzy! [this] seconds remain
	desc:All building production <b>multiplied by 7</b>.
	on tick:
		lose 1 this
		if (this > 0) 
			multiply yield of :Buildings by 7	
		end
	end
	hidden when 0
	class:shinyEffect
*clickFrenzySecond|clickFrenzySeconds 
	name:Click Frenzy!
	text:Click Frenzy! [this] seconds remain
	desc:Drawing fake money <b>x777</b>!.
	on tick:
		lose 1 this
		if (this > 0) 
			multiply yield of drawFakeMoney by 777
		end
	end
	hidden when 0
	class:shinyEffect

*timer
on tick:
	yield 1 timerSeconds
	if (timerSeconds>58)
		yield 1 timerMinutes
		yield -60 timerSeconds
	end
	if (timerMinutes>59)
		yield 1 timerHours
		yield -60 timerMinutes
	end
	if (timerHours>23)
		yield -24 timerHours
		yield 1 timerDays
	end
end
always hidden

	can be negative

*luckypenny
	name:Lucky penny
	frequency:60
	frequency variation:120
	movement:anywhere growShrink
	on click:
		if (chance(20%))
			yield 77 frenzySeconds
			toast The lucky penny activates a <b><#ddc>Frenzy</#></b> for <b>77 seconds!</b>
		else if (chance(4%))
			yield 13 clickFrenzySeconds
			toast The lucky penny activates a <b><#ddc>Click Frenzy</#></b> for <b>13 seconds!</b>
		else
			$amount=(random(3,5)*60*fakeMoney:ps)
			yield $amount fakeMoney
			toast The lucky penny grants you <b>[$amount]</b> fake money!
		end
	end
	class:shinyIcon
	icon:medicons[0,0]
	no text
	class:hasFlares
